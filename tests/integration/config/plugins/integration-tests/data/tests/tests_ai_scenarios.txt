test "AI Scenario - Sol traffic"
	status active
	description "Depart Earth with two escorts and let civilian traffic fly around for a while."
	sequence
		inject "Three Earthly Barges Save"
		call "Load First Savegame"
		call "Depart"
		apply
			"test: steps to wait" = 400
		label "idle"
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "idle"
			"test: steps to wait" > 0
