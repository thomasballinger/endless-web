# Example: replace every missile's guidance with pure pursuit.
guidance "default"
	# Turn straight at the target as fast as the missile can, even if it is
	# behind us (atan2 gives the full -180 to 180 degree bearing).
	if (and has-target has-lock)
		let to-target (- target-position position)
		let bearing (* to-deg (atan2 (cross facing to-target) (dot facing to-target)))
		set turn (max (- max-turn) (min max-turn bearing))
	elif confused
		set turn (* max-turn confusion-direction)
	else
		set turn 0
