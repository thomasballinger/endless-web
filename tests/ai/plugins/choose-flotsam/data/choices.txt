# Example: the game's "flotsam" choice written as a "choose" script, which
# loops over the list of candidates itself, instead of as a "rate" script,
# which rates one candidate at a time (see data/choices.txt). It makes exactly
# the same choices; compare the two to see how each way reads.
choose "flotsam"
	let best-time 600
	for f in flotsam
		let offset (- f.position position)
		let range (length offset)
		let alignment (dot (unit offset) facing)
		if (not (or (> range 800) (and (> range 100) (< alignment .9) (not yours))))
			let time (rendezvous-time offset (- f.velocity velocity) max-velocity)
			if (not (is-nan time))
				set time (+ time (/ (* to-deg (acos (min 1 (max -1 alignment)))) turn-rate))
				if (< time best-time)
					set best-time time
					set chosen f.index
