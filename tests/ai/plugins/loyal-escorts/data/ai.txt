# Example plugin: the player's escorts stay with the flagship instead of
# chasing nearby enemies, unless the player gives them orders. They still
# shoot at anything in range, since only the movement rules are replaced; the
# turn rules come from the "default" rule set.

ai "player escorts"
	movement
		"stop and wait"
			"must recall"
		"stop and wait"
			"stranded with helper"
			not personality derelict
		"follow orders"
		escort
			"has parent"
		independent
