# Example: proportional navigation. Instead of pointing at the target, turn
# so that the line of sight to the target stops rotating; then the missile
# and the target are on a collision course. This needs no target velocity,
# so every homing weapon can use it, but it does need to remember the line of
# sight from the step before.
guidance "proportional navigation" "basic" "throttle control" "leading" "leading and throttle control"
	# The direction to the target on the last step (zero if there was none).
	memory last-sight (vec 0 0)

	if (not has-lock)
		set last-sight (vec 0 0)
		set turn 0
		stop

	let sight (unit (- target-position position))
	# How far off the nose the target is, from -180 to 180 degrees.
	let off-nose (* to-deg (atan2 (cross facing sight) (dot facing sight)))
	# How many degrees the line of sight turned since the last step.
	let sight-rate 0
	if (!= (length last-sight) 0)
		set sight-rate (* to-deg (asin (cross last-sight sight)))
	set last-sight sight

	if (> (abs off-nose) 60)
		# Too far off course to navigate: turn towards the target first.
		set turn (copysign max-turn off-nose)
	else
		# Turn four times as fast as the line of sight, plus a little
		# towards the target so that a target dead ahead is not ignored.
		set turn (clamp (+ (* 4 sight-rate) (* .05 off-nose)) (- max-turn) max-turn)
