# Scripts for weapons whose seekers misjudge the bearing to the target
# ("seeker noise" in a weapon's "homing" attributes, or --bench-seeker-noise
# on the bench). None of them use "leading": each estimates the target's
# velocity from the seeker's fixes, if at all, so they show what noise does
# to a prediction. All four turn by the full angle to where they aim.
#
#   endless-sky --resources . --config tests/integration/config \
#       --guidance-bench "Sidewinder Missile Launcher" --bench-target Hawk \
#       --bench-seeker-noise 3 --bench-script tests/ai/guidance-noise.txt \
#       --bench-guidance pursuit --bench-guidance "raw lead" \
#       --bench-guidance "smoothed lead 0.2s" --bench-guidance "smoothed lead 1s"

# Aim at the latest fix.
guidance "pursuit"
	if (not has-lock)
		set turn (if confused (* turn confusion-direction) 0)
		stop
	let aim (- target-position position)
	let desired (* to-deg (atan2 (cross facing aim) (dot facing aim)))
	set turn (clamp desired (- turn) turn)

# Estimate the target's velocity as the difference between the last two
# fixes, and aim where that puts the target 0.2 seconds (12 steps) from now.
guidance "raw lead"
	memory last-fix (vec 0 0)
	memory has-fix false
	if (not has-lock)
		set has-fix false
		set turn (if confused (* turn confusion-direction) 0)
		stop
	let estimate (if has-fix (- target-position last-fix) (vec 0 0))
	set last-fix target-position
	set has-fix true
	let aim (- (+ target-position (* estimate 12)) position)
	let desired (* to-deg (atan2 (cross facing aim) (dot facing aim)))
	set turn (clamp desired (- turn) turn)

# The same, but move the velocity estimate only a tenth of the way toward each
# new difference, which smooths out the noise but reacts more slowly.
guidance "smoothed lead 0.2s"
	memory last-fix (vec 0 0)
	memory has-fix false
	memory estimate (vec 0 0)
	if (not has-lock)
		set has-fix false
		set turn (if confused (* turn confusion-direction) 0)
		stop
	if has-fix
		set estimate (+ estimate (* .1 (- (- target-position last-fix) estimate)))
	set last-fix target-position
	set has-fix true
	let aim (- (+ target-position (* estimate 12)) position)
	let desired (* to-deg (atan2 (cross facing aim) (dot facing aim)))
	set turn (clamp desired (- turn) turn)

# The same smoothing, aiming a whole second (60 steps) ahead.
guidance "smoothed lead 1s"
	memory last-fix (vec 0 0)
	memory has-fix false
	memory estimate (vec 0 0)
	if (not has-lock)
		set has-fix false
		set turn (if confused (* turn confusion-direction) 0)
		stop
	if has-fix
		set estimate (+ estimate (* .1 (- (- target-position last-fix) estimate)))
	set last-fix target-position
	set has-fix true
	let aim (- (+ target-position (* estimate 60)) position)
	let desired (* to-deg (atan2 (cross facing aim) (dot facing aim)))
	set turn (clamp desired (- turn) turn)
