# Small edits of the game's default "leading and throttle control" script
# (in data/guidance.txt), to show how changing one line changes how a missile
# flies. In each, only the line after "# Changed:" differs from the default.
# Compare them with the default on the guidance bench, without installing a
# plugin:
#
#   endless-sky --resources . --config tests/integration/config \
#       --guidance-bench "Sidewinder Missile Launcher" \
#       --bench-script tests/ai/guidance-edits.txt

# The default turns by asin(cross(facing, direction)), which is 0 for a target
# exactly behind, so it never turns around. Use the full angle instead.
guidance "full turn"
	if has-lock
		let d (- target-position position)
		let direction (unit d)
		let true-velocity (if (!= drag 0) (/ acceleration drag) (length velocity))
		let steps-to-reach (/ (length d) true-velocity)
		if (< (dot direction target-velocity) 0)
			let normal (vec (y direction) (- (x direction)))
			let normal-speed (dot normal target-velocity)
			let closing-speed (sqrt (max 0 (- (* true-velocity true-velocity) (* normal-speed normal-speed))))
			set d (+ (* closing-speed direction) (* normal-speed normal))
		else
			set d (+ d (* steps-to-reach target-velocity))
			set steps-to-reach (/ (length d) true-velocity)
		set direction (unit d)

		# Changed: the full angle to the target, not just its sine, so a target
		# directly behind is 180 degrees away instead of 0.
		let desired-turn (* to-deg (atan2 (cross facing direction) (dot facing direction)))
		if (> (abs desired-turn) turn)
			set turn (copysign turn desired-turn)
		else
			set turn desired-turn

		let steps-to-face (/ desired-turn turn)
		if (> (* steps-to-face 1.5) steps-to-reach)
			set thrust 0
	elif confused
		set turn (* turn confusion-direction)
	else
		set turn 0

# The default cuts its thrust while it still has a long way to turn, so that
# it turns in a tighter circle. Keep thrusting instead.
guidance "full throttle"
	if has-lock
		let d (- target-position position)
		let direction (unit d)
		let true-velocity (if (!= drag 0) (/ acceleration drag) (length velocity))
		let steps-to-reach (/ (length d) true-velocity)
		if (< (dot direction target-velocity) 0)
			let normal (vec (y direction) (- (x direction)))
			let normal-speed (dot normal target-velocity)
			let closing-speed (sqrt (max 0 (- (* true-velocity true-velocity) (* normal-speed normal-speed))))
			set d (+ (* closing-speed direction) (* normal-speed normal))
		else
			set d (+ d (* steps-to-reach target-velocity))
			set steps-to-reach (/ (length d) true-velocity)
		set direction (unit d)

		let desired-turn (* to-deg (asin (cross facing direction)))
		if (> (abs desired-turn) turn)
			set turn (copysign turn desired-turn)
		else
			set turn desired-turn

		let steps-to-face (/ desired-turn turn)
		# Changed: never cut the engine to turn more tightly.
		if false
			set thrust 0
	elif confused
		set turn (* turn confusion-direction)
	else
		set turn 0

# Aim straight at the target instead of leading it.
guidance "no lead"
	if has-lock
		let d (- target-position position)
		let direction (unit d)
		let true-velocity (if (!= drag 0) (/ acceleration drag) (length velocity))
		let steps-to-reach (/ (length d) true-velocity)
		if (< (dot direction target-velocity) 0)
			let normal (vec (y direction) (- (x direction)))
			let normal-speed (dot normal target-velocity)
			let closing-speed (sqrt (max 0 (- (* true-velocity true-velocity) (* normal-speed normal-speed))))
			set d (+ (* closing-speed direction) (* normal-speed normal))
		else
			set d (+ d (* steps-to-reach target-velocity))
			set steps-to-reach (/ (length d) true-velocity)
		# Changed: aim at where the target is, not where it will be.
		set direction (unit (- target-position position))

		let desired-turn (* to-deg (asin (cross facing direction)))
		if (> (abs desired-turn) turn)
			set turn (copysign turn desired-turn)
		else
			set turn desired-turn

		let steps-to-face (/ desired-turn turn)
		if (> (* steps-to-face 1.5) steps-to-reach)
			set thrust 0
	elif confused
		set turn (* turn confusion-direction)
	else
		set turn 0
