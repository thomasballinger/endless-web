# program's script is the "guidance" or "maneuver" node of the same name.

# Weave from side to side on the way in, and straighten out at the end.
guidance "Corkscrew"
	if target
		turn = turn_toward(target.direction)
		if target.distance > 300
			turn = turn + max_turn * .7 * sin(age / 5)
	else
		turn = 0

# Steer to stop the line of sight to the target from turning.
guidance "Proportional Navigation"
	if target
		if abs(target.bearing) > 40
			# Too far off for that: face the target first.
			turn = turn_toward(target.intercept)
		else
			# How fast the line of sight is turning, in radians per step, and
			# a little of the bearing, to straighten out at the end.
			let relative = target.velocity - velocity
			let sight_rate = cross(target.offset, relative) / (target.distance * target.distance)
			turn = 4 * sight_rate * to_deg + .1 * target.bearing
	else
		turn = 0

# Come in from one side, at an angle that shrinks as it closes.
guidance "Flanker"
	memory side = 0
	if side == 0
		side = random() < .5 ? -1 : 1
	if target
		let angle = side * 30 * clamp((target.distance - 300) / 900, 0, 1) * to_rad
		let d = target.direction
		let aim = vec(d.x * cos(angle) - d.y * sin(angle), d.x * sin(angle) + d.y * cos(angle))
		turn = turn_toward(aim)
	else
		turn = 0

# Coast until facing the target, then burn.
guidance "Patient Strike"
	if target
		turn = turn_toward(target.direction)
		if abs(target.bearing) > 30
			thrust = 0
	else
		turn = 0
		thrust = 0

# Turn gently.
guidance "Thrifty Seeker"
	if target
		turn = turn_toward(target.direction) * .6
	else
		turn = 0

# Remember where the target was, to work out how it is moving, and lead it.
guidance "Lead Without Leading"
	memory last_seen = vec(0, 0)
	memory have_last = false
	if target
		let moving = have_last ? target.position - last_seen : vec(0, 0)
		last_seen = target.position
		have_last = true
		turn = turn_toward(unit(target.offset + moving * target.steps_away))
	else
		have_last = false
		turn = 0

# Hold position 250 behind the leader and 200 to its right.
maneuver "Wingman (Right)"
	let right = vec(-target_facing.y, target_facing.x)
	let station = target_position - target_facing * 250 + right * 200
	let offset = station - position
	# Close in faster the farther off station, up to half top speed.
	let approach = unit(offset) * min(max_velocity * .5, length(offset) * .02)
	let change = target_velocity + approach - velocity
	# Face the change to make, or once it is small, the way the leader faces.
	let goal = length(change) > acceleration * 2 ? change : target_facing
	let angle = heading(goal) - facing_degrees
	if abs(angle) > 180
		angle = angle + (angle < 0 ? 360 : -360)
	turn = clamp(angle / turn_rate, -1, 1)
	if length(change) > acceleration * 2 && dot(facing, unit(change)) > .9
		forward = true

# Fly 350 ahead of the leader.
maneuver "Vanguard"
	let station = target_position + target_facing * 350
	let offset = station - position
	let approach = unit(offset) * min(max_velocity * .5, length(offset) * .02)
	let change = target_velocity + approach - velocity
	let goal = length(change) > acceleration * 2 ? change : target_facing
	let angle = heading(goal) - facing_degrees
	if abs(angle) > 180
		angle = angle + (angle < 0 ? 360 : -360)
	turn = clamp(angle / turn_rate, -1, 1)
	if length(change) > acceleration * 2 && dot(facing, unit(change)) > .9
		forward = true

# Tuck in 120 behind the leader.
maneuver "Close Guard"
	let station = target_position - target_facing * 120
	let offset = station - position
	let approach = unit(offset) * min(max_velocity * .5, length(offset) * .03)
	let change = target_velocity + approach - velocity
	let goal = length(change) > acceleration * 2 ? change : target_facing
	let angle = heading(goal) - facing_degrees
	if abs(angle) > 180
		angle = angle + (angle < 0 ? 360 : -360)
	turn = clamp(angle / turn_rate, -1, 1)
	if length(change) > acceleration * 2 && dot(facing, unit(change)) > .9
		forward = true
