# Maneuver scripts fly a ship relative to another object for one step. They are
# written in the same language as guidance scripts (see guidance.txt), with
# these inputs:
#   position velocity facing facing-degrees     this ship (facing is a unit vector)
#   target-position target-velocity target-facing
#   max-velocity acceleration turn-rate inertial-mass drag-force reverse-thrust
#   last-turn                                   the turn command given last step
#   should-use-afterburner                      whether it has the fuel, energy
#                                               and cooling for its afterburner
#   top-speed                                   max-velocity, counting the
#                                               afterburner if it should use it
#   radius slow cruise-speed                    for "move to" (see below)
#   command-turn                                the turn command so far this step
#   aim                                         for "attack": the direction to point
#                                               the guns to hit the target
#   ramming target-max-velocity artillery       for "fight" (see below)
#   shortest-range shortest-artillery min-safe-distance
#   ship-radius target-radius max-reverse-velocity reverse-acceleration
# and these outputs (starting at command-turn or false):
#   turn          how far to turn, in units of the ship's turn rate
#   forward       whether to thrust forward
#   back          whether to thrust in reverse
#   afterburner   whether to use the afterburner
#   arrived       for "move to": whether the ship is there
# and these actions, which only "fight" uses:
#   attack        fly the "attack" maneuver after this one
#   aim-guns      only turn to aim the guns at the target
# and functions from the game:
#   (rendezvous-time offset velocity speed)     steps to reach a moving point
#   (heading v)                                 direction of v in degrees
# and, only for "move to", "pick up", "attack" and "fight", three about the
# ship itself:
#   (stopping-point target-velocity)            where it would come to rest
#   (stopping-reverse target-velocity)          whether reverse thrust stops it sooner
#   (turn-toward v)                             the turn command that points it along v

# Hold position near a moving target, matching its velocity and then its
# facing. Escorts use this to stay with their parent. This reproduces the
# game's built-in behavior exactly.
maneuver "keep station"
	let max-time 600
	let lead-time 500
	let position-deadband 200
	let velocity-deadband 1.5
	let time-deadband 120
	let thrust-deadband .5

	# Where to be, relative to where the ship is now.
	let velocity-delta (- target-velocity velocity)
	let position-delta (- (+ target-position (* lead-time velocity-delta)) position)
	let position-size (length position-delta)
	let position-weight (/ position-size (+ position-size position-deadband))
	# How fast to be going, relative to how fast the ship is going now.
	set velocity-delta (- velocity-delta (* facing velocity-deadband))
	let velocity-size (length velocity-delta)
	let velocity-weight (/ velocity-size (+ velocity-size velocity-deadband))

	# Roughly how long it will take to get to the target.
	let position-time (rendezvous-time position-delta target-velocity max-velocity)
	if (or (is-nan position-time) (> position-time max-time))
		set position-time max-time
	let rendezvous (+ position-delta (* target-velocity position-time))
	let position-turn (abs (- facing-degrees (heading rendezvous)))
	set position-time (+ position-time (/ (min position-turn (- 360 position-turn)) turn-rate))
	set position-time (+ position-time (/ (length (- (* (unit rendezvous) max-velocity) velocity)) acceleration))
	# If very close, stop trying to adjust.
	set position-time (* position-time (* position-weight position-weight))

	# Roughly how long it will take to match the target's velocity.
	let velocity-time (/ (length velocity-delta) acceleration)
	let velocity-turn (abs (- facing-degrees (heading velocity-delta)))
	set velocity-time (+ velocity-time (/ (min velocity-turn (- 360 velocity-turn)) turn-rate))
	set velocity-time (* velocity-time (* velocity-weight velocity-weight))

	# Focus on matching position or velocity, whichever will take longer.
	let total-time (+ position-time velocity-time time-deadband)
	set position-weight (/ position-time total-time)
	set velocity-weight (/ velocity-time total-time)
	let facing-weight (/ time-deadband total-time)

	# Face somewhere between the rendezvous point, the velocity change, and
	# the target's own facing.
	let facing-goal (+ (* (unit rendezvous) position-weight) (* (unit velocity-delta) velocity-weight) (* target-facing facing-weight))
	let target-angle (- (heading facing-goal) facing-degrees)
	if (> (abs target-angle) 180)
		set target-angle (+ target-angle (if (< target-angle 0) 360 -360))
	# Avoid "turn jitter" when position and velocity are well matched.
	let changed-direction (!= (signbit last-turn) (signbit target-angle))
	let target-turn (abs (/ target-angle turn-rate))
	let previous-turn (abs last-turn)
	if (and (!= previous-turn 0) (or changed-direction (and (< previous-turn 1) (> target-turn previous-turn))))
		# Keep turning the same way, but limit how fast the turn speeds up.
		let damped-turn (+ (if changed-direction 0 previous-turn) (min .025 target-turn))
		set turn (copysign damped-turn target-angle)
	elif (< target-turn 1)
		set turn (copysign target-turn target-angle)
	else
		set turn target-angle

	# Decide whether to thrust. Drag is left out for reverse thrust, because
	# the estimate of how it applies can be quite inaccurate.
	let drag (* velocity drag-force)
	if (!= reverse-thrust 0)
		let reverse (unit (* facing (/ (- reverse-thrust) inertial-mass)))
		let direction (+ (/ (* position-weight (dot position-delta reverse)) position-deadband) (/ (* velocity-weight (dot velocity-delta reverse)) velocity-deadband))
		if (> direction thrust-deadband)
			set back true
			stop
	let ahead (unit (- (* facing acceleration) drag))
	let direction (+ (/ (* position-weight (dot position-delta ahead)) position-deadband) (/ (* velocity-weight (dot velocity-delta ahead)) velocity-deadband))
	if (> direction thrust-deadband)
		set forward true

# Stay close to a target without matching its velocity: point at it, and fly
# toward it when it is more than 200 units away and ahead of the ship (using
# the afterburner beyond 750). Ships use this to hang around a ship they are
# attacking, guarding or waiting for. This reproduces the game's built-in
# behavior exactly.
maneuver "circle around"
	let direction (- target-position position)

	# Turn toward the target. Inside one step's turn of the target direction,
	# stop turning once the facing is within about 0.8 degrees of it.
	let precision .9999
	let side (cross direction facing)
	let ahead (dot direction facing)
	let turning-toward false
	if (> ahead 0)
		let close (and (< precision 1) (> precision 0) (>= (* ahead ahead) (* precision (dot direction direction))))
		let angle (* (asin (min 1 (max -1 (/ side (length direction))))) to-deg)
		if (< (abs angle) turn-rate)
			set turning-toward true
			if close
				set turn 0
			else
				set turn (/ (- angle) turn-rate)
	if (not turning-toward)
		set turn (if (< side 0) 1 -1)

	let distance (length direction)
	if (and (> distance 200) (>= (dot facing direction) 0))
		set forward true
		if (and (> distance 750) should-use-afterburner)
			set afterburner true

# Fly to target-position and arrive with target-velocity: within radius and
# slower than slow, relative to it. With a cruise speed, hold that speed on the
# way. The AI uses this for many things: approaching planets to land, mining,
# swarming, patrolling and more. This reproduces the game's built-in behavior
# exactly.
maneuver "move to"
	let offset (- target-position position)
	let relative-speed (length (- target-velocity velocity))
	let is-close (< (length offset) radius)
	if (and is-close (< relative-speed slow))
		set arrived true
		stop

	# Aim at where the ship would stop, not where it is.
	set offset (- target-position (stopping-point target-velocity))
	let should-reverse (stopping-reverse target-velocity)

	# The direction to accelerate in to get where the ship wants to be.
	let aim offset
	let has-cruise-speed (> cruise-speed 0)
	if has-cruise-speed
		# Prefer to fly toward the target at cruise speed. If already close to
		# that velocity, face the target.
		set aim (- (* (unit offset) cruise-speed) velocity)
		if (< (dot aim aim) .01)
			set aim offset

	let is-facing (> (dot (unit aim) facing) .95)
	if (or (not is-close) (and (not is-facing) (not should-reverse)))
		set turn (turn-toward aim)

	# Drag only applies while thrusting, so stop thrusting just below top
	# speed, unless the course needs correcting.
	let limit (* top-speed .99)
	if (and is-facing (or (<= (dot velocity velocity) (* limit limit)) (< (dot (unit offset) (unit velocity)) .95)))
		let heading-there (> (dot (unit velocity) (unit offset)) .95)
		if (or (not has-cruise-speed) (not heading-there) (< (length velocity) cruise-speed))
			set forward true
	elif should-reverse
		set turn (turn-toward velocity)
		set back true

# Fly into a piece of flotsam to collect it: aim where it will be by the time
# the ship gets there, thrust when roughly facing it, and use the afterburner
# when that won't make the ship overshoot. This reproduces the game's built-in
# behavior exactly.
maneuver "pick up"
	# Where the flotsam is, and how it moves, relative to the ship.
	let offset (- target-position position)
	let drift (- target-velocity velocity)

	# Estimate where it will be by the time the ship reaches it.
	let time (rendezvous-time offset drift max-velocity)
	if (is-nan time)
		set time (/ (length offset) max-velocity)
	let degrees-to-turn (* to-deg (acos (min 1 (max -1 (dot (unit offset) facing)))))
	set time (+ time (/ degrees-to-turn turn-rate))
	set offset (+ offset (* drift time))

	set turn (turn-toward offset)
	let alignment (dot (unit offset) facing)
	if (> alignment .7)
		set forward true

	# The closer the flotsam, the more precisely the ship must face it before
	# using the afterburner.
	let square-distance (dot offset offset)
	# (The game's code also checks whether forward thrust was commanded before
	# this maneuver ran, but facing this closely means forward is set anyway.)
	if (and forward should-use-afterburner)
		if (> alignment (max .9 (min .9999 (- 1 (/ square-distance 10000000)))))
			set afterburner true

# Close in on a target while pointing the guns at it. Ships use this to fight
# other ships and to mine asteroids. This reproduces the game's built-in
# behavior exactly.
maneuver "attack"
	set turn (turn-toward aim)

	# The smallest circle the ship can turn in at its current speed.
	let steps-in-full-turn (/ 360 turn-rate)
	let circumference (* steps-in-full-turn (length velocity))
	let diameter (max 200 (/ circumference pi))

	let direction (- target-position position)
	let alignment (dot facing (unit direction))
	# With reverse thrusters, back toward a target behind the ship.
	if (and (< alignment -.75) (!= reverse-thrust 0))
		set back true
	# Thrust when facing the target and far enough away not to overshoot while
	# still turning, or when moving away from it but facing it.
	elif (or (and (>= alignment 0) (> (length direction) diameter)) (and (< (dot velocity direction) 0) (>= alignment .9)))
		set forward true
		if (and (> (length direction) 600) should-use-afterburner)
			set afterburner true

# Decide how to fight a ship. A ship that rams always closes in. A ship whose
# weapons could hurt it up close, or that fights with long-range artillery,
# keeps its distance if it is fast enough to; otherwise it closes in until
# its shortest-range weapon reaches and then just aims. Inputs:
#   ramming               whether its personality is to ram its target
#   target-max-velocity   the target's top speed
#   artillery             whether it fights with mostly long-range weapons
#   shortest-range        the range of its shortest-range weapon
#   shortest-artillery    the range of its shortest-range artillery
#   min-safe-distance     how far to stay from the target to be safe from its
#                         own weapons' blasts
#   ship-radius target-radius
#   max-reverse-velocity reverse-acceleration
# This reproduces the game's built-in behavior exactly.
maneuver "fight"
	if ramming
		attack
		stop

	# Keeping away only works with a 10% edge in speed over the target.
	let can-run (< (* target-max-velocity 1.1) max-velocity)
	let use-artillery (and artillery can-run)
	let safe-distance (if can-run min-safe-distance 0)

	let total-radius (+ ship-radius target-radius)
	let direction (- target-position position)
	# Roughly how far the ship's weapons are from the target's hull.
	let weapon-distance (- (length direction) (/ total-radius 3))

	if (or (!= safe-distance 0) (and use-artillery (< shortest-range weapon-distance)))
		set safe-distance (+ (* 1.25 safe-distance) total-radius)
		let approach-speed (dot (- velocity target-velocity) (unit direction))
		# Back away with reverse thrusters if they are fast enough.
		let use-reverse (and (!= max-reverse-velocity 0) (or (>= max-reverse-velocity (min target-max-velocity max-velocity)) (<= (dot target-velocity (- (unit direction))) max-reverse-velocity)))
		let slowdown-distance (/ (/ (* approach-speed approach-speed) (if use-reverse reverse-acceleration (+ acceleration (/ 160 turn-rate)))) 2)

		# Too close: get away.
		if (< (length direction) (max (+ safe-distance (max slowdown-distance 0)) (* (if use-artillery 1 0) .75 shortest-artillery)))
			if use-reverse
				set turn (turn-toward direction)
				if (>= (dot facing direction) 0)
					set back true
			else
				set turn (turn-toward (- direction))
				if (<= (dot facing direction) 0)
					set forward true
		elif (or (and use-artillery (and (> approach-speed 0) (< weapon-distance (* shortest-artillery .9)))) (< weapon-distance (* shortest-range .75)))
			aim-guns
		else
			attack
	# Fire if in range, or else move closer.
	elif (< weapon-distance (* shortest-range .75))
		aim-guns
	else
		attack
