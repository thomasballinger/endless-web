# Maneuver scripts fly a ship relative to another object for one step. They are
# written in the same language as guidance scripts (see guidance.txt), with
# these inputs:
#   position velocity facing facing-degrees     this ship (facing is a unit vector)
#   target-position target-velocity target-facing
#   max-velocity acceleration turn-rate inertial-mass drag-force reverse-thrust
#   last-turn                                   the turn command given last step
#   should-use-afterburner                      whether it has the fuel, energy
#                                               and cooling for its afterburner
# and these outputs (starting at 0 or false):
#   turn          how far to turn, in units of the ship's turn rate
#   forward       whether to thrust forward
#   back          whether to thrust in reverse
#   afterburner   whether to use the afterburner
# and two functions from the game:
#   (rendezvous-time offset velocity speed)     steps to reach a moving point
#   (heading v)                                 direction of v in degrees

# Hold position near a moving target, matching its velocity and then its
# facing. Escorts use this to stay with their parent. This reproduces the
# game's built-in behavior exactly.
maneuver "keep station"
	let max-time 600
	let lead-time 500
	let position-deadband 200
	let velocity-deadband 1.5
	let time-deadband 120
	let thrust-deadband .5

	# Where to be, relative to where the ship is now.
	let velocity-delta (- target-velocity velocity)
	let position-delta (- (+ target-position (* lead-time velocity-delta)) position)
	let position-size (length position-delta)
	let position-weight (/ position-size (+ position-size position-deadband))
	# How fast to be going, relative to how fast the ship is going now.
	set velocity-delta (- velocity-delta (* facing velocity-deadband))
	let velocity-size (length velocity-delta)
	let velocity-weight (/ velocity-size (+ velocity-size velocity-deadband))

	# Roughly how long it will take to get to the target.
	let position-time (rendezvous-time position-delta target-velocity max-velocity)
	if (or (is-nan position-time) (> position-time max-time))
		set position-time max-time
	let rendezvous (+ position-delta (* target-velocity position-time))
	let position-turn (abs (- facing-degrees (heading rendezvous)))
	set position-time (+ position-time (/ (min position-turn (- 360 position-turn)) turn-rate))
	set position-time (+ position-time (/ (length (- (* (unit rendezvous) max-velocity) velocity)) acceleration))
	# If very close, stop trying to adjust.
	set position-time (* position-time (* position-weight position-weight))

	# Roughly how long it will take to match the target's velocity.
	let velocity-time (/ (length velocity-delta) acceleration)
	let velocity-turn (abs (- facing-degrees (heading velocity-delta)))
	set velocity-time (+ velocity-time (/ (min velocity-turn (- 360 velocity-turn)) turn-rate))
	set velocity-time (* velocity-time (* velocity-weight velocity-weight))

	# Focus on matching position or velocity, whichever will take longer.
	let total-time (+ position-time velocity-time time-deadband)
	set position-weight (/ position-time total-time)
	set velocity-weight (/ velocity-time total-time)
	let facing-weight (/ time-deadband total-time)

	# Face somewhere between the rendezvous point, the velocity change, and
	# the target's own facing.
	let facing-goal (+ (* (unit rendezvous) position-weight) (* (unit velocity-delta) velocity-weight) (* target-facing facing-weight))
	let target-angle (- (heading facing-goal) facing-degrees)
	if (> (abs target-angle) 180)
		set target-angle (+ target-angle (if (< target-angle 0) 360 -360))
	# Avoid "turn jitter" when position and velocity are well matched.
	let changed-direction (!= (signbit last-turn) (signbit target-angle))
	let target-turn (abs (/ target-angle turn-rate))
	let previous-turn (abs last-turn)
	if (and (!= previous-turn 0) (or changed-direction (and (< previous-turn 1) (> target-turn previous-turn))))
		# Keep turning the same way, but limit how fast the turn speeds up.
		let damped-turn (+ (if changed-direction 0 previous-turn) (min .025 target-turn))
		set turn (copysign damped-turn target-angle)
	elif (< target-turn 1)
		set turn (copysign target-turn target-angle)
	else
		set turn target-angle

	# Decide whether to thrust. Drag is left out for reverse thrust, because
	# the estimate of how it applies can be quite inaccurate.
	let drag (* velocity drag-force)
	if (!= reverse-thrust 0)
		let reverse (unit (* facing (/ (- reverse-thrust) inertial-mass)))
		let direction (+ (/ (* position-weight (dot position-delta reverse)) position-deadband) (/ (* velocity-weight (dot velocity-delta reverse)) velocity-deadband))
		if (> direction thrust-deadband)
			set back true
			stop
	let ahead (unit (- (* facing acceleration) drag))
	let direction (+ (/ (* position-weight (dot position-delta ahead)) position-deadband) (/ (* velocity-weight (dot velocity-delta ahead)) velocity-deadband))
	if (> direction thrust-deadband)
		set forward true

# Stay close to a target without matching its velocity: point at it, and fly
# toward it when it is more than 200 units away and ahead of the ship (using
# the afterburner beyond 750). Ships use this to hang around a ship they are
# attacking, guarding or waiting for. This reproduces the game's built-in
# behavior exactly.
maneuver "circle around"
	let direction (- target-position position)

	# Turn toward the target. Inside one step's turn of the target direction,
	# stop turning once the facing is within about 0.8 degrees of it.
	let precision .9999
	let side (cross direction facing)
	let ahead (dot direction facing)
	let turning-toward false
	if (> ahead 0)
		let close (and (< precision 1) (> precision 0) (>= (* ahead ahead) (* precision (dot direction direction))))
		let angle (* (asin (min 1 (max -1 (/ side (length direction))))) to-deg)
		if (< (abs angle) turn-rate)
			set turning-toward true
			if close
				set turn 0
			else
				set turn (/ (- angle) turn-rate)
	if (not turning-toward)
		set turn (if (< side 0) 1 -1)

	let distance (length direction)
	if (and (> distance 200) (>= (dot facing direction) 0))
		set forward true
		if (and (> distance 750) should-use-afterburner)
			set afterburner true
